cmake_minimum_required(VERSION 3.16)
project(openarm_smolvla LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(std_msgs REQUIRED)
set(VR_APP_ROOT "$ENV{HOME}/Desktop/vive-focus-vision-ros2" CACHE PATH "VR app checkout supplying external controller interface v1")
if(NOT EXISTS "${VR_APP_ROOT}/include/vive_vr_ros2/external_control.hpp")
  message(FATAL_ERROR "Set VR_APP_ROOT to the updated VR app checkout")
endif()
add_library(openarm_smolvla_control SHARED src/guarded_policy.cpp)
target_include_directories(openarm_smolvla_control PRIVATE include "${VR_APP_ROOT}/include")
ament_target_dependencies(openarm_smolvla_control rclcpp trajectory_msgs std_srvs std_msgs)
add_executable(policy_guard_check tests/policy_guard_check.cpp)
target_include_directories(policy_guard_check PRIVATE include "${VR_APP_ROOT}/include")
enable_testing()
add_test(NAME policy_guard_check COMMAND policy_guard_check)
